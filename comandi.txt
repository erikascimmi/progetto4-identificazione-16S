# Progetto 5 — Comandi utilizzati

## Download dati grezzi (SRA Toolkit)
cd ~/Desktop/progetto4-identificazione-16S

prefetch SRR36601846

fasterq-dump SRR36601846 --split-files -O dati/

## Trimming (Trimmomatic)
java -jar trimmomatic-0.39.jar PE dati/SRR36601846_1.fastq dati/SRR36601846_2.fastq \
  dati/SRR36601846_1_paired.fastq dati/SRR36601846_1_unpaired.fastq \
  dati/SRR36601846_2_paired.fastq dati/SRR36601846_2_unpaired.fastq \
  LEADING:3 TRAILING:3 SLIDINGWINDOW:4:20 MINLEN:100

## Merge
vsearch --fastq_mergepairs dati/SRR36601846_1_paired.fastq \
  --reverse dati/SRR36601846_2_paired.fastq \
  --fastqout dati/SRR36601846_merged.fastq \
  --fastq_minovlen 20 --fastq_maxdiffs 5

seqtk seq -a dati/SRR36601846_merged.fastq > dati/SRR36601846_merged.fasta

## Estrazione sequenza rappresentativa
head -n 2 dati/SRR36601846_merged.fasta > dati/SRR36601846_una_sequenza.fasta
